# 缩放点积注意力

```python
import torch
import math

def scaled_dot_product_attention(query: torch.Tensor, key: torch.Tensor, value: torch.Tensor, mask: torch.Tensor = None)->tuple[torch.Tensor, torch.Tensor]:
    """
    计算缩放点积注意力。

    参数：
        query: 查询张量；形状为 (batch_size, num_heads, seq_len_q, d_k)
               或 (batch_size, seq_len_q, d_k) (如果为单头)。
        key: 键张量；形状为 (batch_size, num_heads, seq_len_k, d_k)
             或 (batch_size, seq_len_k, d_k) (如果为单头)。
        value: 值张量；形状为 (batch_size, num_heads, seq_len_v, d_v)
               或 (batch_size, seq_len_v, d_v) (如果为单头)。
               注意：seq_len_k 和 seq_len_v 必须相同。
        mask: 可选的掩码张量；形状应可广播到
              (batch_size, num_heads, seq_len_q, seq_len_k)。
              `True` 或 `1` 的位置将被掩盖（设为 -inf）。

    返回：
        包含以下内容的元组：
        - output：注意力输出张量；
                  形状为 (batch_size, num_heads, seq_len_q, d_v)
                  或 (batch_size, seq_len_q, d_v) (如果为单头)。
        - attention_weights：注意力权重张量；
                             形状为 (batch_size, num_heads, seq_len_q, seq_len_k)
                             或 (batch_size, seq_len_q, seq_len_k) (如果为单头)。
    """

    # 获取键向量的最后一维大小（d_k）
    d_k = query.size(-1)
    # 计算缩放点积：(Q @ K^T) / sqrt(d_k)
    scores = torch.matmul(query, key.transpose(-2, -1)) / math.sqrt(d_k)

    # 如果提供了掩码，则将掩码位置设为负无穷
    if mask is not None:
        scores = scores.masked_fill(mask, float('-inf'))

    # 对分数应用softmax以获得注意力权重，dim=-1表示最后一个维度，即seq_len_k，每一行
    attention_weights = torch.nn.functional.softmax(scores, dim=-1)

    # 处理NaN值，将其替换为0.0
    if torch.isnan(attention_weights).any():
        attention_weights = torch.nan_to_num(attention_weights, nan=0.0)

    # 使用注意力权重加权求和值向量
    output = torch.matmul(attention_weights, value)
    # 返回输出和注意力权重
    return output, attention_weights

# 示例用法

seq_len_k = 7

d_v = 128

query_tensor = torch.randn(batch_size, seq_len_q, d_k)
key_tensor = torch.randn(batch_size, seq_len_k, d_k)
value_tensor = torch.randn(batch_size, seq_len_k, d_v)

# 掩码示例
padding_mask = torch.zeros(batch_size, 1,seq_len_k, dtype=torch.bool)
padding_mask[:, :, -2:] = True

# 调用缩放点积注意力函数
output, attention_weights = scaled_dot_product_attention(query_tensor, key_tensor, value_tensor, padding_mask)

print("Output shape:", output.shape)
print("attention weights shape:", attention_weights.shape)


print(attention_weights)
```

# 多头注意力

```python
import torch
import math

def scaled_dot_product_attention(query: torch.Tensor, key: torch.Tensor, value: torch.Tensor, mask: torch.Tensor = None)->tuple[torch.Tensor, torch.Tensor]:
    """
    计算缩放点积注意力。

    参数：
        query: 查询张量；形状为 (batch_size, num_heads, seq_len_q, d_k)
               或 (batch_size, seq_len_q, d_k) (如果为单头)。
        key: 键张量；形状为 (batch_size, num_heads, seq_len_k, d_k)
             或 (batch_size, seq_len_k, d_k) (如果为单头)。
        value: 值张量；形状为 (batch_size, num_heads, seq_len_v, d_v)
               或 (batch_size, seq_len_v, d_v) (如果为单头)。
               注意：seq_len_k 和 seq_len_v 必须相同。
        mask: 可选的掩码张量；形状应可广播到
              (batch_size, num_heads, seq_len_q, seq_len_k)。
              `True` 或 `1` 的位置将被掩盖（设为 -inf）。

    返回：
        包含以下内容的元组：
        - output：注意力输出张量；
                  形状为 (batch_size, num_heads, seq_len_q, d_v)
                  或 (batch_size, seq_len_q, d_v) (如果为单头)。
        - attention_weights：注意力权重张量；
                             形状为 (batch_size, num_heads, seq_len_q, seq_len_k)
                             或 (batch_size, seq_len_q, seq_len_k) (如果为单头)。
    """

    # 获取键向量的最后一维大小（d_k）
    d_k = query.size(-1)
    # 计算缩放点积：(Q @ K^T) / sqrt(d_k)
    scores = torch.matmul(query, key.transpose(-2, -1)) / math.sqrt(d_k)

    # 如果提供了掩码，则将掩码位置设为负无穷
    if mask is not None:
        scores = scores.masked_fill(mask, float('-inf'))

    # 对分数应用softmax以获得注意力权重，dim=-1表示最后一个维度，即seq_len_k，每一行
    attention_weights = torch.nn.functional.softmax(scores, dim=-1)

    # 处理NaN值，将其替换为0.0
    if torch.isnan(attention_weights).any():
        attention_weights = torch.nan_to_num(attention_weights, nan=0.0)

    # 使用注意力权重加权求和值向量
    output = torch.matmul(attention_weights, value)
    # 返回输出和注意力权重
    return output, attention_weights


class MultiHeadAttention(torch.nn.Module):
    # 定义多头注意力类，继承自torch.nn.Module
    def __init__(self, d_model: int, num_heads: int, dropout: float = 0.1):
        """
        初始化多头注意力模块。

        参数：
            d_model: 模型的隐藏维度。
            num_heads: 注意力头的数量。
            dropout: Dropout的概率。
        """

        # 调用父类的初始化方法
        super().__init__()

        # 断言d_model能被num_heads整除
        assert d_model % num_heads == 0, "d_model must be divisible by num_heads"

        # 存储模型隐藏维度
        self.d_model = d_model
        # 存储注意力头的数量
        self.num_heads = num_heads

        # 计算每个头的维度
        self.d_k = d_model // num_heads

        # 创建查询的线性变换层
        self.W_q = torch.nn.Linear(d_model, d_model)
        # 创建键的线性变换层
        self.W_k = torch.nn.Linear(d_model, d_model)
        # 创建值的线性变换层
        self.W_v = torch.nn.Linear(d_model, d_model)

        # 创建输出的线性变换层
        self.W_o = torch.nn.Linear(d_model, d_model)
        # 创建Dropout层
        self.dropout = torch.nn.Dropout(dropout)  # Dropout层（可选）

    def split_heads(self, x: torch.Tensor) -> torch.Tensor:
        """
        将输入张量拆分为多头。

        参数：
            x: 输入张量，形状为 (batch_size, seq_len, d_model)

        返回：
            拆分后的张量，形状为 (batch_size, num_heads, seq_len, d_k)
        """
        # 获取批量大小和序列长度
        batch_size, seq_len, _ = x.size()
        # 重新调整张量形状为 (batch_size, seq_len, num_heads, d_k)
        x = x.view(batch_size, seq_len, self.num_heads, self.d_k)
        # 转置第1和2维，得到 (batch_size, num_heads, seq_len, d_k)
        return x.transpose(1, 2)  # (batch_size, num_heads, seq_len, d_k)

    def combine_heads(self, x: torch.Tensor) -> torch.Tensor:
        """
        将多头张量合并为单头。

        参数：
            x: 多头张量，形状为 (batch_size, num_heads, seq_len, d_k)

        返回：
            合并后的张量，形状为 (batch_size, seq_len, d_model)
        """
        # 获取多头张量的维度信息
        batch_size, num_heads, seq_len, d_k = x.size()
        # 转置第1和2维得到 (batch_size, seq_len, num_heads, d_k)，并保证内存连续
        x = x.transpose(1, 2).contiguous()  # (batch_size, seq_len, num_heads, d_k)
        # 重新调整张量形状为 (batch_size, seq_len, d_model)
        return x.view(batch_size, seq_len, num_heads * d_k)  # (batch_size, seq_len, d_model)

    def forward(self, query: torch.Tensor, key: torch.Tensor, value: torch.Tensor, mask: torch.Tensor = None) -> torch.Tensor:
        """
        多头注意力的前向传播。

        参数：
            query: 查询张量，形状为 (batch_size, seq_len_q, d_model)
            key: 键张量，形状为 (batch_size, seq_len_k, d_model)
            value: 值张量，形状为 (batch_size, seq_len_k, d_model)
            mask: 可选的掩码张量，形状为 (batch_size, 1, seq_len_k) 或 (batch_size, seq_len_q, seq_len_k)

        返回：
            输出张量，形状为 (batch_size, seq_len_q, d_model)
            注意力权重，形状为 (batch_size, num_heads, seq_len_q, seq_len_k)
        """
        # 对查询进行线性变换
        Q = self.W_q(query)
        # 对键进行线性变换
        K = self.W_k(key)
        # 对值进行线性变换
        V = self.W_v(value)

        # 将查询拆分为多个头
        Q = self.split_heads(Q)
        # 将键拆分为多个头
        K = self.split_heads(K)
        # 将值拆分为多个头
        V = self.split_heads(V)

        # 计算缩放点积注意力并获得输出和权重
        output, attention_weights = scaled_dot_product_attention(Q, K, V, mask)

        # 将多个头的输出合并为单头
        output = self.combine_heads(output)

        # 对合并后的输出进行线性变换
        output = self.W_o(output)
        # 应用Dropout正则化
        output = self.dropout(output)

        # 返回最终输出和注意力权重
        return output, attention_weights

batch_size = 2
seq_len = 10
d_model = 512
num_heads = 8

x = torch.rand(batch_size, seq_len, d_model)

mha = MultiHeadAttention(d_model, num_heads)
output, attention_weights = mha(x, x, x)

print("input shape:", x.shape)
print("output shape:", output.shape)
print("attention weights shape:", attention_weights.shape)
```

疑问：Q = self.W_q(query) 进行切分哪里没搞懂，这样不是每个头都是用的同一个线性变化么？没有体现一个头一个query 权重w么？

解答：

原论文（Attention is All You Need）里写的就是这种形式：

$$
\text{head}_i = \text{Attention}(QW_i^Q, KW_i^K, VW_i^V)
$$
其中每个 \(W_i^Q\) 的形状是 \(d_{\text{model}} \times d_k\)。如果把所有 head 的投影矩阵水平拼起来：



$$
W^Q = \begin{bmatrix} W_1^Q & W_2^Q & \cdots & W_h^Q \end{bmatrix}
\quad (d_{\text{model}} \times d_{\text{model}})
$$
然后一次矩阵乘法：

```python
Q = X @ W^Q          # 得到 (batch, seq, d_model)
```

其实与公式多头计算是一致的，这里这样一次大矩阵乘法是方便加快计算。
